func person(actor: StringName) -> VisitState:
 var v:=s.visitors.spawn(actor);v.pose.ground=s.player_ground+Vector2(0,-0.2);v.destination=v.pose.ground;v.phase=&"staying";v.shopping_state=&"declined";return v
