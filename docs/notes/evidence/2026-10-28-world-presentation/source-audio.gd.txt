extends SceneTree
var content: ContentStore
var session: GameSession
var locale: LocaleService
func _initialize() -> void:
 call_deferred("observe")
func observe() -> void:
 root.size=Vector2i(1280,720)
 content=ContentStore.new()
 root.add_child(content)
 content.required_sections=SessionConfig.REQUIRED_SECTIONS.duplicate()
 var loaded := content.load_directory("res://content")
 print("CONTENT ",loaded.ok," ",loaded.details)
 if not loaded.ok:quit(1);return
 locale=LocaleService.new()
 locale.configure(content.bundle())
 session=GameSession.new()
 root.add_child(session)
 var configured := session.configure(SessionConfig.new(content.bundle()))
 print("SESSION ",configured.ok," ",configured.details)
 if not configured.ok:quit(1);return
 var sound := Soundscape.new()
 root.add_child(sound)
 sound.configure(content.section("atmosphere").audio)
 for room in ["kiosk","rooftop"]:
  sound.present(StringName(room),session.cooking,session.home,session.items,session.spaces,session.workflow)
  print("IDLE ",room," ",sound._targets)
 session.location_id=&"kiosk"
 session.player_ground=session.spaces.ground(&"tea")
 var tea := session.workflow.start(&"tea",&"kiosk",PackedStringArray(["tea"]))
 print("TEA START ",tea.ok," ",tea.details)
 sound.present(&"kiosk",session.cooking,session.home,session.items,session.spaces,session.workflow)
 print("TEA LOCAL ",sound._targets)
 sound.present(&"rooftop",session.cooking,session.home,session.items,session.spaces,session.workflow)
 print("TEA REMOTE ",sound._targets)
 sound.set_muted(true)
 sound.present(&"kiosk",session.cooking,session.home,session.items,session.spaces,session.workflow)
 print("MUTED ",sound._targets)
 for id in sound._players:
  print("ASSET ",id," ",sound._players[id].stream.resource_path," seconds ",sound._players[id].stream.get_length())
 sound.free()
 session.free()
 content.free()
 quit()
